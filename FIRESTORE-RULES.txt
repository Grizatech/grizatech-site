rules_version = '2';


service cloud.firestore {
  match /databases/{database}/documents {


    function isAdmin() {
      return request.auth != null
        && request.auth.uid == 'G3c1Rt5AtKfBaJKEJ8w9qdKQHLA2';
    }


    match /reviews/{reviewId} {


      // O público vê somente avaliações aprovadas.
      // O administrador pode ver todas.
      allow read: if resource.data.status == 'approved'
        || isAdmin();


      // Qualquer visitante pode enviar uma nova avaliação,
      // mas obrigatoriamente como pendente.
      allow create: if
        request.resource.data.keys().hasOnly([
          'name',
          'service',
          'rating',
          'comment',
          'status',
          'createdAt'
        ])
        && request.resource.data.keys().hasAll([
          'name',
          'service',
          'rating',
          'comment',
          'status',
          'createdAt'
        ])
        && request.resource.data.status == 'pending'
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5
        && request.resource.data.createdAt == request.time;


      // Só você pode alterar, aprovar, ocultar ou excluir.
      allow update, delete: if isAdmin();
    }
    match /galleryNames/{photoId} {
      allow read: if true;
      allow create, update: if isAdmin()
        && photoId.matches('^(0[1-9]|[1-9][0-9])$')
        && request.resource.data.keys().hasOnly(['name', 'updatedAt'])
        && request.resource.data.keys().hasAll(['name', 'updatedAt'])
        && request.resource.data.name is string
        && request.resource.data.name.size() <= 60
        && request.resource.data.updatedAt == request.time;
      allow delete: if isAdmin();
    }
  }
}
